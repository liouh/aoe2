// find first
__aoe2rec.operations.find((item) => {
    if (item.Action && item.Action.action_data && item.Action.action_data.Build) {
        return true;
    }
});

// find all
__aoe2rec.operations.filter((item) => {
    if (item.Action && item.Action.action_data && item.Action.action_data.DeQueue) {
        console.log(item.Action.action_data);
    }
});

// find all by player
__aoe2rec.operations.filter((item) => {
    if (item.Action && item.Action.action_data && item.Action.action_data.DeQueue
        && item.Action.action_data[Object.keys(item.Action.action_data)[0]].player_id === 1) {
        console.log(item.Action.action_data);
    }
});

// count actions for one player
let actionCounts = {};
__aoe2rec.operations.forEach((item) => {
    const actionData = item.Action?.action_data;
    if (actionData) {
        const actionType = Object.keys(actionData)[0];
        if (actionData[actionType]?.player_id === 1) {
            actionCounts[actionType] = (actionCounts[actionType] || 0) + 1;
        }
    }
});
console.table(actionCounts);

// count actions by player
let playerActionStats = {};
__aoe2rec.operations.forEach((item) => {
    const actionData = item.Action?.action_data;
    if (actionData) {
        const actionType = Object.keys(actionData)[0];
        const playerId = actionData[actionType]?.player_id;
        if (playerId !== undefined) {
            if (!playerActionStats[playerId]) {
                playerActionStats[playerId] = {};
            }
            playerActionStats[playerId][actionType] = (playerActionStats[playerId][actionType] || 0) + 1;
        }
    }
});
console.table(playerActionStats);
